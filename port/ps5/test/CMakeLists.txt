cmake_minimum_required(VERSION 3.25)
project(ico_ps5_host_tests C)
set(CMAKE_C_STANDARD 11)
find_package(SDL3 CONFIG REQUIRED COMPONENTS SDL3-static)
enable_testing()
foreach(name input audio)
    add_executable(ps5_${name}_test ${name}_test.c)
    target_link_libraries(ps5_${name}_test PRIVATE SDL3::SDL3-static)
    add_test(NAME ps5_${name} COMMAND ps5_${name}_test)
    set_tests_properties(ps5_${name} PROPERTIES TIMEOUT 5)
endforeach()
